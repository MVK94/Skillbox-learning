Для списко вможно пременять различные предустановленные методы

например есть список langs = ['Python', 'Java', 'JS', 'SQL']

если использовать метод index, получим индекс ячейки
i_lang = langs.index('Java')
i_lang = 1

если использовать метод insert, вставим где хотим
langs.insert(i_lang + 1, 'C++')
вставит после Java на 2 ячейку

если использовать метод remove, удалим что хотим
langs.remove('JS')

метод extend (дословно Расширить)
если нужно элементы одного списка перенести в другой список

my_list = ['Игра', 'Изгой', 'Таксист']
your_list = ['Начало', 'Отступники', 'Король Лев']


my_list.extend(your_list)

my_list = ['Игра', 'Изгой', 'Таксист', 'Начало', 'Отступники', 'Король Лев']

my_list.extend('слово')
my_list = ['Игра', 'Изгой', 'Таксист', 'Начало', 'Отступники', 'Король Лев','с','л','о','в','о']

метод count подсчитывает количество элементов в списке
decode = [1,1,0,1,0, -1]
k = decode.count(1)
#k == 3
